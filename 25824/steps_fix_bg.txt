Друг потребител отваря реда и вижда статуса. Запис на Genehmigt или Abgelehnt не минава. Смяна към `cancelled` не се спира. Genehmigung, която не е за `sn_quote_mgmt_core_quote`, не се пипа.

## Стъпка 1. Script Include

1. Превключи приложението на Global.
2. `sys_script_include.list`, New.
3. Name: `QuoteApprovalGuard`
4. Client callable: изключено
5. Скрипт:

```javascript
var QuoteApprovalGuard = Class.create();
QuoteApprovalGuard.prototype = {
    initialize: function() {},

    QUOTE_TABLE: 'sn_quote_mgmt_core_quote',

    isQuoteApproval: function(rec) {
        if (!rec)
            return false;
        if (rec.getValue('source_table') == this.QUOTE_TABLE)
            return true;
        var taskId = rec.getValue('sysapproval');
        if (!taskId)
            return false;
        var task = new GlideRecord('task');
        if (!task.get(taskId))
            return false;
        return task.getValue('sys_class_name') == this.QUOTE_TABLE;
    },

    canDecide: function(rec) {
        if (!rec)
            return false;
        if (gs.getUserName() == 'system')
            return true;
        if (gs.hasRole('admin') || gs.hasRole('approval_admin'))
            return true;
        var approverId = rec.getValue('approver');
        if (!approverId)
            return false;
        if (approverId == gs.getUserID())
            return true;
        return this._isActiveApprovalDelegate(approverId);
    },

    canWriteState: function(rec) {
        if (!this.isQuoteApproval(rec))
            return true;
        var state = rec.getValue('state');
        if (state != 'approved' && state != 'rejected')
            return true;
        return this.canDecide(this._stored(rec));
    },

    canWriteApprover: function(rec) {
        if (!this.isQuoteApproval(rec))
            return true;
        return this.canDecide(this._stored(rec));
    },

    blockUnauthorizedDecision: function(current, previous) {
        if (!this.isQuoteApproval(current))
            return;
        if (this.canDecide(previous))
            return;

        var newState = current.getValue('state');
        var setsOutcome = current.state.changes() && (newState == 'approved' || newState == 'rejected');
        var movesApprover = current.approver.changes();
        if (!setsOutcome && !movesApprover)
            return;

        gs.addErrorMessage('Nur der eingetragene Genehmiger darf diese Genehmigung genehmigen oder ablehnen.');
        current.setAbortAction(true);
    },

    _stored: function(rec) {
        var stored = new GlideRecord('sysapproval_approver');
        if (stored.get(rec.getUniqueValue()))
            return stored;
        return rec;
    },

    _isActiveApprovalDelegate: function(approverId) {
        var now = new GlideDateTime();
        var del = new GlideRecord('sys_user_delegate');
        del.addQuery('user', approverId);
        del.addQuery('delegate', gs.getUserID());
        del.addQuery('approvals', true);
        var startQuery = del.addNullQuery('starts');
        startQuery.addOrCondition('starts', '<=', now);
        var endQuery = del.addNullQuery('ends');
        endQuery.addOrCondition('ends', '>=', now);
        del.setLimit(1);
        del.query();
        return del.hasNext();
    },

    type: 'QuoteApprovalGuard'
};
```

6. Submit.

`canWriteState` и `canWriteApprover` гледат Genehmiger, който вече е в базата. Иначе заявката първо сменя Genehmiger на себе си и после минава проверката.

## Стъпка 5. Полетата на формата са само за четене

ACL на `state` пуска полето, докато стойността още е Angefordert, и отказва чак при запис на Genehmigt. Без това правило падащото меню остава активно. UI policy го заключва веднага за човек, който не е Genehmiger.

В операторите няма `is not (dynamic)`. Има само `is (dynamic)`, най-отдолу в списъка. На немски е `ist (dynamisch)`. След него третото поле е списък, не търсене на потребител. Избери `Me` (`Ich`).

Правят се две policy. Първата заключва полетата на всяка Quote Genehmigung. Втората ги отключва само когато Genehmiger е влезлият потребител. По-големият Order печели.
